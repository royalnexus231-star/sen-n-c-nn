import math import random import sys import pygame # Pygame Başlatma pygame.init() # Ekran Boyutları WIDTH, HEIGHT = 800, 600 screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("Python Heart Code") # Renkler ve Font BLACK = (0, 0, 0) WHITE = (255, 255, 255) font = pygame.font.SysFont("arial", 12, bold=True) text_surface = font.render("I Love You", True, WHITE) # Kalp Matematiksel Formülü def heart_function(t): x = 16 * (math.sin(t) ** 3) y = -( 13 * math.cos(t) - 5 * math.cos(2 * t) - 2 * math.cos(3 * t) - math.cos(4 * t) ) return x, y # Parçacık (Particle) Sınıfı class Particle: def __init__(self): t = random.uniform(0, 2 * math.pi) hx, hy = heart_function(t) # Ekranın merkezine ölçekleme scale = 15 self.target_x = WIDTH // 2 + hx * scale self.target_y = HEIGHT // 2 + hy * scale # Rastgele başlangıç pozisyonu self.x = random.randint(0, WIDTH) self.y = random.randint(0, HEIGHT) self.delay = random.uniform(0, 50) self.alpha = 0 self.flicker = random.uniform(0.5, 1.5) def update(self, frame): if frame > self.delay: # Hedefe doğru yavaşça hareket etme (Lerp) self.x += (self.target_x - self.x) * 0.05 self.y += (self.target_y - self.y) * 0.05 # Şeffaflığı artırma if self.alpha < 255: self.alpha = min(255, self.alpha + 5) # Parıltı/titreşim efekti self.current_alpha = min( 255, max( 0, int( self.alpha * (0.75 + 0.25 * math.sin(frame * 0.1 * self.flicker)) ), ), ) else: self.current_alpha = 0 def draw(self, surface): if self.current_alpha > 0: temp_surface = text_surface.copy() temp_surface.set_alpha(self.current_alpha) surface.blit(temp_surface, (self.x, self.y)) # Parçacıkları Oluşturma num_particles = 600 particles = [Particle() for _ in range(num_particles)] clock = pygame.time.Clock() frame = 0 running = True # Ana Döngü while running: for event in pygame.event.get(): if event.type == pygame.QUIT or ( event.type == pygame.KEYDOWN and event.key == pygame.K_ESCAPE ): running = False screen.fill(BLACK) for p in particles: p.update(frame) p.draw(screen) pygame.display.flip() frame += 1 clock.tick(60) pygame.quit() sys.exit()
